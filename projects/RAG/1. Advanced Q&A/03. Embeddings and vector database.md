from functools import lru_cache

from sentence_transformers import SentenceTransformer

from app.core.config import settings


class EmbeddingService:
    def __init__(self) -> None:
        self.model = SentenceTransformer(
            settings.embedding_model
        )

    def embed_text(self, text: str) -> list[float]:
        vector = self.model.encode(
            text,
            normalize_embeddings=True,
        )

        return vector.tolist()

    def embed_texts(
        self,
        texts: list[str],
    ) -> list[list[float]]:
        vectors = self.model.encode(
            texts,
            normalize_embeddings=True,
        )

        return vectors.tolist()


@lru_cache
def get_embedding_service() -> EmbeddingService:
    return EmbeddingService()

def create_collection_if_not_exists() -> None:
    client = get_qdrant_client()

# 4.8 Why cosine similarity?

Our embedding vectors are normalized:

```python
normalize_embeddings=True
```

Therefore cosine similarity is a natural choice.

Conceptually:

```text
                 vector A
                    ↘
                     ↘
                      angle
                     ↗
                    ↗
                 vector B

small angle
    ↓
high similarity
```

Qdrant performs the similarity search for us.

# 4.9 Qdrant repository

Create:

```text
app/qdrant/repository.py
```

```python
from uuid import UUID

from qdrant_client.models import PointStruct

from app.core.config import settings
from app.qdrant.client import get_qdrant_client


class QdrantRepository:
    def __init__(self) -> None:
        self.client = get_qdrant_client()
        self.collection_name = settings.qdrant_collection_name

    def upsert_chunk(
        self,
        chunk_id: UUID,
        document_id: UUID,
        vector: list[float],
        text: str,
        chunk_index: int,
        page_number: int | None,
    ) -> None:

        point = PointStruct(
            id=str(chunk_id),
            vector=vector,
            payload={
                "chunk_id": str(chunk_id),
                "document_id": str(document_id),
                "chunk_index": chunk_index,
                "page_number": page_number,
                "text": text,
            },
        )

        self.client.upsert(
            collection_name=self.collection_name,
            points=[point],
        )

    def upsert_chunks(
        self,
        points: list[PointStruct],
    ) -> None:

        if not points:
            return

        self.client.upsert(
            collection_name=self.collection_name,
            points=points,
        )

    def delete_document_chunks(
        self,
        document_id: UUID,
    ) -> None:

        self.client.delete(
            collection_name=self.collection_name,
            points_selector={
                "filter": {
                    "must": [
                        {
                            "key": "document_id",
                            "match": {
                                "value": str(document_id),
                            },
                        }
                    ]
                }
            },
        )
```

# 4.10 Vector service
VectorService = "Take this document's chunks, turn them into vectors, package them as Qdrant points, and index them in Qdrant."
Now connect:

```text
PostgreSQL
    ↓
DocumentChunk
    ↓
EmbeddingService
    ↓
QdrantRepository
```

Create:

```text
app/services/vector_service.py
```

```python
from qdrant_client.models import PointStruct
from sqlalchemy import select
from sqlalchemy.orm import Session

from app.embeddings.service import get_embedding_service
from app.models.document_chunk import DocumentChunk
from app.qdrant.repository import QdrantRepository


class VectorService:
    def __init__(self) -> None:
        self.embedding_service = get_embedding_service()
        self.qdrant_repository = QdrantRepository()

    def index_document(
        self,
        db: Session,
        document_id,
    ) -> int:

        chunks = db.scalars(
            select(DocumentChunk)
            .where(
                DocumentChunk.document_id == document_id
            )
            .order_by(DocumentChunk.chunk_index)
        ).all()

        if not chunks:
            return 0

        texts = [
            chunk.text
            for chunk in chunks
        ]

        vectors = self.embedding_service.embed_texts(
            texts
        )

        points = []

        for chunk, vector in zip(
            chunks,
            vectors,
            strict=True,
        ):
            points.append(
                PointStruct(
                    id=str(chunk.id),
                    vector=vector,
                    payload={
                        "chunk_id": str(chunk.id),
                        "document_id": str(chunk.document_id),
                        "chunk_index": chunk.chunk_index,
                        "page_number": chunk.page_number,
                        "text": chunk.text,
                    },
                )
            )

        self.qdrant_repository.upsert_chunks(
            points
        )

        return len(points)
```

# 4.16 Qdrant data structure

A chunk in PostgreSQL:

```text
DocumentChunk
────────────────────────────
id
document_id
chunk_index
text
page_number
token_count
created_at
```

Corresponding Qdrant point:

```text
Point
────────────────────────────
id
vector[384]

payload
├── chunk_id
├── document_id
├── chunk_index
├── page_number
└── text
```

Example:

```json
{
  "id": "chunk-uuid",
  "vector": [0.021, -0.114, 0.083, "..."],
  "payload": {
    "chunk_id": "chunk-uuid",
    "document_id": "document-uuid",
    "chunk_index": 4,
    "page_number": 2,
    "text": "FastAPI is a modern Python web framework..."
  }
}
```
